Ejercicio 1: Ocurrencias y Eliminación en Lista

Se lee una secuencia de números que se almacena en una lista encadenada. Además se lee otro número por teclado. Se desea: a) Determinar cuántas veces aparece ese número en la lista. b) Eliminarlo si aparece una sola vez, si no existe imprimir un mensaje indicando este hecho.

#include <stdio.h>
#include <stdlib.h>

struct Nodo {
    int dato;
    struct Nodo* sig;
};

void insertar(struct Nodo** cabeza, int valor) {
    struct Nodo* nuevo = (struct Nodo*)malloc(sizeof(struct Nodo));
    nuevo->dato = valor;
    nuevo->sig = *cabeza;
    *cabeza = nuevo;
}

int contarOcurrencias(struct Nodo* cabeza, int buscado) {
    int contador = 0;
    struct Nodo* actual = cabeza;
    while (actual != NULL) {
        if (actual->dato == buscado) contador++;
        actual = actual->sig;
    }
    return contador;
}

void eliminarUnico(struct Nodo** cabeza, int buscado) {
    struct Nodo *actual = *cabeza, *anterior = NULL;
    
    // Si la cabeza es el nodo a eliminar
    if (actual != NULL && actual->dato == buscado) {
        *cabeza = actual->sig;
        free(actual);
        printf("Elemento %d eliminado con exito (estaba en la cabeza).\n", buscado);
        return;
    }
    
    // Buscar el nodo
    while (actual != NULL && actual->dato != buscado) {
        anterior = actual;
        actual = actual->sig;
    }
    
    // Desvincular y liberar
    if (actual != NULL) {
        anterior->sig = actual->sig;
        free(actual);
        printf("Elemento %d eliminado con exito.\n", buscado);
    }
}

void procesarLista(struct Nodo** cabeza, int buscado) {
    int cant = contarOcurrencias(*cabeza, buscado);
    
    if (cant == 0) {
        printf("El numero %d NO existe en la lista.\n", buscado);
    } else {
        printf("El numero %d aparece %d vez/veces en la lista.\n", buscado, cant);
        if (cant == 1) {
            eliminarUnico(cabeza, buscado);
        } else {
            printf("Aparece mas de una vez, no se elimina.\n");
        }
    }
}

int main() {
    struct Nodo* lista = NULL;
    int nro_buscado = 15;
    
    // Simulacion de carga
    insertar(&lista, 10);
    insertar(&lista, 15);
    insertar(&lista, 20);
    insertar(&lista, 30);
    
    procesarLista(&lista, nro_buscado); // Aparece 1 vez (se eliminara)
    procesarLista(&lista, 99);          // No existe
    
    return 0;
}

Ejercicio 2: Biblioteca Escolar (Inserción Ordenada)

En una escuela primaria se ha pedido a los alumnos que donen un libro. Se desea: a) Incorporar los datos a una lista en forma ordenada. (Título, género, autor, tema, editorial). b) Informar la cantidad de fábulas y cuentos infantiles donados.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Libro {
    char titulo[50];
    char genero[30];
    char autor[30];
    char tema[30];
    char editorial[30];
};

struct NodoLibro {
    struct Libro dato;
    struct NodoLibro* sig;
};

// Insercion ordenada alfabeticamente por Titulo
void insertarOrdenado(struct NodoLibro** cabeza, struct Libro nuevo_libro) {
    struct NodoLibro* nuevo = (struct NodoLibro*)malloc(sizeof(struct NodoLibro));
    nuevo->dato = nuevo_libro;
    nuevo->sig = NULL;
    
    if (*cabeza == NULL || strcmp((*cabeza)->dato.titulo, nuevo_libro.titulo) > 0) {
        nuevo->sig = *cabeza;
        *cabeza = nuevo;
    } else {
        struct NodoLibro* actual = *cabeza;
        while (actual->sig != NULL && strcmp(actual->sig->dato.titulo, nuevo_libro.titulo) < 0) {
            actual = actual->sig;
        }
        nuevo->sig = actual->sig;
        actual->sig = nuevo;
    }
}

void reporteBiblioteca(struct NodoLibro* cabeza) {
    int fabulas = 0, infantiles = 0;
    struct NodoLibro* actual = cabeza;
    
    printf("--- LISTADO DE LIBROS ORDENADO ---\n");
    while (actual != NULL) {
        printf("Titulo: %s | Genero: %s\n", actual->dato.titulo, actual->dato.genero);
        
        // Uso estricto en minusculas para la comparacion
        if (strcmp(actual->dato.genero, "fabula") == 0) fabulas++;
        if (strcmp(actual->dato.genero, "infantil") == 0) infantiles++;
        
        actual = actual->sig;
    }
    
    printf("\n--- ESTADISTICAS ---\n");
    printf("Cantidad de Fabulas: %d\n", fabulas);
    printf("Cantidad de Cuentos Infantiles: %d\n", infantiles);
}

int main() {
    struct NodoLibro* biblioteca = NULL;
    
    struct Libro l1 = {"Zorro y Uvas", "fabula", "Esopo", "Moraleja", "Ed1"};
    struct Libro l2 = {"Caperucita", "infantil", "Perrault", "Bosque", "Ed2"};
    struct Libro l3 = {"Alicia", "infantil", "Carroll", "Fantasia", "Ed3"};
    struct Libro l4 = {"El Principito", "fabula", "Exupery", "Filosofia", "Ed4"};
    
    insertarOrdenado(&biblioteca, l1);
    insertarOrdenado(&biblioteca, l2);
    insertarOrdenado(&biblioteca, l3);
    insertarOrdenado(&biblioteca, l4);
    
    reporteBiblioteca(biblioteca);
    
    return 0;
}

Ejercicio 3: Comparación de Listas de Supermercados

Se tiene una lista de precios de Carrefour y otra con los precios de los mismos productos en Disco. Se desea obtener una tercera lista con los nombres de cada producto y el nombre del supermercado donde conviene comprarlo.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Producto {
    char nombre[30];
    float precio;
};

struct NodoProd {
    struct Producto dato;
    struct NodoProd* sig;
};

struct Resultado {
    char nombre[30];
    char supermercado[20];
};

struct NodoRes {
    struct Resultado dato;
    struct NodoRes* sig;
};

void insertarProd(struct NodoProd** cabeza, char nom[], float pre) {
    struct NodoProd* nuevo = (struct NodoProd*)malloc(sizeof(struct NodoProd));
    strcpy(nuevo->dato.nombre, nom);
    nuevo->dato.precio = pre;
    nuevo->sig = *cabeza;
    *cabeza = nuevo;
}

void insertarRes(struct NodoRes** cabeza, char nom[], char super[]) {
    struct NodoRes* nuevo = (struct NodoRes*)malloc(sizeof(struct NodoRes));
    strcpy(nuevo->dato.nombre, nom);
    strcpy(nuevo->dato.supermercado, super);
    nuevo->sig = *cabeza;
    *cabeza = nuevo;
}

// Se asume que ambas listas tienen los mismos productos
void generarListaOptima(struct NodoProd* carrefour, struct NodoProd* disco, struct NodoRes** resultado) {
    struct NodoProd *actualC = carrefour;
    
    while (actualC != NULL) {
        struct NodoProd *actualD = disco;
        
        // Buscar el producto homonimo en Disco
        while (actualD != NULL && strcmp(actualC->dato.nombre, actualD->dato.nombre) != 0) {
            actualD = actualD->sig;
        }
        
        if (actualD != NULL) {
            if (actualC->dato.precio < actualD->dato.precio) {
                insertarRes(resultado, actualC->dato.nombre, "Carrefour");
            } else if (actualD->dato.precio < actualC->dato.precio) {
                insertarRes(resultado, actualC->dato.nombre, "Disco");
            } else {
                insertarRes(resultado, actualC->dato.nombre, "Indistinto");
            }
        }
        actualC = actualC->sig;
    }
}

void imprimirResultados(struct NodoRes* cabeza) {
    struct NodoRes* actual = cabeza;
    printf("--- LISTA OPTIMA DE COMPRAS ---\n");
    while (actual != NULL) {
        printf("Producto: %-15s | Conviene en: %s\n", actual->dato.nombre, actual->dato.supermercado);
        actual = actual->sig;
    }
}

int main() {
    struct NodoProd *listaCarrefour = NULL;
    struct NodoProd *listaDisco = NULL;
    struct NodoRes *listaFinal = NULL;
    
    insertarProd(&listaCarrefour, "Leche", 500.0);
    insertarProd(&listaCarrefour, "Pan", 800.0);
    
    insertarProd(&listaDisco, "Leche", 480.0);
    insertarProd(&listaDisco, "Pan", 850.0);
    
    generarListaOptima(listaCarrefour, listaDisco, &listaFinal);
    imprimirResultados(listaFinal);
    
    return 0;
}

Ejercicio 4: Inmobiliaria (Ordenación Indirecta)

De cada departamento se conoce: ubicación, dormitorios, piso, cochera, antigüedad, precio y expensas. Informar cuál es el más caro y generar un informe con el total de departamentos según cantidad de dormitorios, ordenados de mayor a menor por dicho total.

Se cuenta la cantidad de departamentos según número de dormitorios usando un pequeño vector estático auxiliar, y luego se ordena ese vector mediante el algoritmo de Burbuja para cumplir el requerimiento del reporte.

#include <stdio.h>
#include <stdlib.h>

struct Depto {
    char ubicacion[30];
    int dormitorios;
    int piso;
    int cochera; // 1 si, 0 no
    int antiguedad;
    float precio;
    float expensas;
};

struct NodoDepto {
    struct Depto dato;
    struct NodoDepto* sig;
};

void insertarDepto(struct NodoDepto** cabeza, struct Depto d) {
    struct NodoDepto* nuevo = (struct NodoDepto*)malloc(sizeof(struct NodoDepto));
    nuevo->dato = d;
    nuevo->sig = *cabeza;
    *cabeza = nuevo;
}

void analizarInmobiliaria(struct NodoDepto* cabeza) {
    if (cabeza == NULL) return;
    
    struct NodoDepto* actual = cabeza;
    float max_precio = actual->dato.precio;
    
    // Arreglo para contar deptos por cantidad de dormitorios (indice = cant. dorm)
    // Suponemos maximo 10 dormitorios para el ejemplo
    int conteo_dorm[10] = {0}; 
    int indices_dorm[10]; // Para no perder que indice (dormitorio) corresponde tras ordenar
    for(int i=0; i<10; i++) indices_dorm[i] = i;
    
    while (actual != NULL) {
        if (actual->dato.precio > max_precio) {
            max_precio = actual->dato.precio;
        }
        if (actual->dato.dormitorios < 10) {
            conteo_dorm[actual->dato.dormitorios]++;
        }
        actual = actual->sig;
    }
    
    printf("El departamento mas caro cuesta: $%.2f\n\n", max_precio);
    
    // Ordenar arreglos paralelamente (Burbuja Descendente por conteo)
    for(int i = 0; i < 9; i++) {
        for(int j = 0; j < 9 - i; j++) {
            if(conteo_dorm[j] < conteo_dorm[j+1]) {
                int aux_c = conteo_dorm[j];
                conteo_dorm[j] = conteo_dorm[j+1];
                conteo_dorm[j+1] = aux_c;
                
                int aux_i = indices_dorm[j];
                indices_dorm[j] = indices_dorm[j+1];
                indices_dorm[j+1] = aux_i;
            }
        }
    }
    
    printf("--- AGRUPACION POR DORMITORIOS (MAYOR A MENOR) ---\n");
    for(int i = 0; i < 10; i++) {
        if(conteo_dorm[i] > 0) {
            printf("%d Dormitorios: %d departamentos\n", indices_dorm[i], conteo_dorm[i]);
        }
    }
}

int main() {
    struct NodoDepto* lista = NULL;
    
    struct Depto d1 = {"Centro", 2, 3, 1, 5, 80000.0, 5000.0};
    struct Depto d2 = {"Norte", 3, 1, 1, 10, 120000.0, 7000.0};
    struct Depto d3 = {"Sur", 2, 5, 0, 2, 75000.0, 4500.0};
    struct Depto d4 = {"Centro", 1, 2, 0, 1, 60000.0, 3000.0};
    
    insertarDepto(&lista, d1);
    insertarDepto(&lista, d2);
    insertarDepto(&lista, d3);
    insertarDepto(&lista, d4);
    
    analizarInmobiliaria(lista);
    
    return 0;
}

De un vivero se almacena: código, nombre, precio, stock, ubicación (interior/exterior) y tipo (árbol/arbusto/flor). a) Generar e imprimir otra lista con arbustos de exterior. b) Calcular porcentaje de plantas de exterior e interior.

#include <stdio.h>

#include <string.h>

struct Planta {

void procesarVivero(struct NodoPlanta* original, struct NodoPlanta** sublista) {
    int total = 0, cant_ext = 0, cant_int = 0;
    struct NodoPlanta* actual = original;
    
    while (actual != NULL) {
        total++;
        
        if (actual->dato.exterior == 1) cant_ext++;
        else cant_int++;
        
        // Si es Arbusto de Exterior, generar sublista
        if (actual->dato.exterior == 1 && strcmp(actual->dato.tipo, "arbusto") == 0) {
            insertar(sublista, actual->dato);

        actual = actual->sig;
    }
    
    printf("--- ESTADISTICAS DEL VIVERO ---\n");
    if (total > 0) {

Ejercicio 1: Videoclub (Lista de Socios con Historial)

Simular el proceso de alquiler de películas. Cada socio registra sus alquileres (tipo, titulo, etc). Al final del día: a) Cantidad total de películas/juegos alquilados y monto. b) Detalle por socio con su historial (títulos), ordenado por código de socio de menor a mayor.

Dado que no existe un límite fijo de alquileres por socio, utilizamos una "Lista de Listas": Una lista principal ordenada de Socios, y dentro de cada Socio, un puntero a su propia lista enlazada de Alquileres.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

// Sub-lista: Historial de alquileres de un socio
struct Alquiler {
    int tipo; // 1: Pelicula, 2: Juego
    char titulo[30];
    float precio;
};

struct NodoAlquiler {
    struct Alquiler dato;
    struct NodoAlquiler* sig;
};

// Lista Principal: Socios (Ordenada por Codigo)
struct Socio {
    int cod_socio;
    struct NodoAlquiler* historial; // Puntero a la sublista
};

struct NodoSocio {
    struct Socio dato;
    struct NodoSocio* sig;
};

void registrarAlquiler(struct NodoSocio** listaSocios, int cod_socio, struct Alquiler alq, int* tot_p, int* tot_j, float* monto) {
    // Actualizar metricas globales
    if (alq.tipo == 1) (*tot_p)++; else (*tot_j)++;
    *monto += alq.precio;

    // Buscar socio (o insertar si no existe, manteniendo orden)
    struct NodoSocio* actual = *listaSocios;
    struct NodoSocio* anterior = NULL;
    
    while (actual != NULL && actual->dato.cod_socio < cod_socio) {
        anterior = actual;
        actual = actual->sig;
    }
    
    // Si no existe, crearlo
    if (actual == NULL || actual->dato.cod_socio != cod_socio) {
        struct NodoSocio* nuevoSocio = (struct NodoSocio*)malloc(sizeof(struct NodoSocio));
        nuevoSocio->dato.cod_socio = cod_socio;
        nuevoSocio->dato.historial = NULL; // Lista de alquileres vacia
        nuevoSocio->sig = actual;
        
        if (anterior == NULL) *listaSocios = nuevoSocio;
        else anterior->sig = nuevoSocio;
        
        actual = nuevoSocio;
    }
    
    // Insertar alquiler en el historial del socio
    struct NodoAlquiler* nuevoAlq = (struct NodoAlquiler*)malloc(sizeof(struct NodoAlquiler));
    nuevoAlq->dato = alq;
    nuevoAlq->sig = actual->dato.historial;
    actual->dato.historial = nuevoAlq;
}

void emitirReporte(struct NodoSocio* cabeza, int tp, int tj, float monto) {
    printf("--- REPORTE GLOBAL DEL DIA ---\n");
    printf("Peliculas Alquiladas: %d | Juegos Alquilados: %d\n", tp, tj);
    printf("Monto Recaudado: $%.2f\n\n", monto);
    
    printf("--- DETALLE POR SOCIO (Ordenado por Codigo) ---\n");
    struct NodoSocio* actual = cabeza;
    while (actual != NULL) {
        int c_p = 0, c_j = 0;
        printf("Socio Nro: %d\n", actual->dato.cod_socio);
        
        struct NodoAlquiler* h = actual->dato.historial;
        printf("  Historial:\n");
        while (h != NULL) {
            printf("    - [%s] %s\n", (h->dato.tipo == 1) ? "Peli" : "Juego", h->dato.titulo);
            if (h->dato.tipo == 1) c_p++; else c_j++;
            h = h->sig;
        }
        printf("  Total Socio: %d Peliculas, %d Juegos.\n\n", c_p, c_j);
        actual = actual->sig;
    }
}

int main() {
    struct NodoSocio* sistema = NULL;
    int total_pelis = 0, total_juegos = 0;
    float monto_total = 0.0;
    
    struct Alquiler a1 = {1, "Matrix", 500.0};
    struct Alquiler a2 = {2, "FIFA 25", 800.0};
    struct Alquiler a3 = {1, "Inception", 500.0};
    
    // Registros (Simulacion de atencion en el mostrador)
    registrarAlquiler(&sistema, 150, a1, &total_pelis, &total_juegos, &monto_total);
    registrarAlquiler(&sistema, 102, a2, &total_pelis, &total_juegos, &monto_total);
    registrarAlquiler(&sistema, 150, a3, &total_pelis, &total_juegos, &monto_total);
    
    emitirReporte(sistema, total_pelis, total_juegos, monto_total);
    
    return 0;
}

Ejercicio 2: Competencia (Arreglo de Listas)

Competencia en 10 disciplinas. Alumnos se inscriben en múltiples pruebas. El día del evento se registran, compiten y obtienen puntaje. Se pide simular el proceso para imprimir los 3 mejores por disciplina, y estadísticas (inscriptos vs presentes por prueba y asistencia por escuela).

Al ser exactamente 10 disciplinas fijas, la estructura idónea es un Arreglo de Listas. Se declara un vector de tamaño 10, donde cada celda es el puntero inicial a la lista enlazada de competidores inscritos en esa disciplina.

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define DISCIPLINAS 3 // Usaremos 3 para el ejemplo practico (el PDF indica 10)

struct Participante {
    int dni;
    char escuela[30];
    int presente; // 1: Asistio, 0: Ausente
    float puntaje;
};

struct NodoComp {
    struct Participante dato;
    struct NodoComp* sig;
};

void inscribir(struct NodoComp** disciplina, struct Participante p) {
    p.presente = 0; // Por defecto ausente al inscribirse por internet
    p.puntaje = 0.0;
    
    struct NodoComp* nuevo = (struct NodoComp*)malloc(sizeof(struct NodoComp));
    nuevo->dato = p;
    nuevo->sig = *disciplina;
    *disciplina = nuevo;
}

void registrarPresencia(struct NodoComp* disciplina, int dni) {
    struct NodoComp* actual = disciplina;
    while (actual != NULL) {
        if (actual->dato.dni == dni) {
            actual->dato.presente = 1;
            return;
        }
        actual = actual->sig;
    }
}

void cargarPuntaje(struct NodoComp* disciplina, int dni, float pts) {
    struct NodoComp* actual = disciplina;
    while (actual != NULL) {
        if (actual->dato.dni == dni && actual->dato.presente == 1) {
            actual->dato.puntaje = pts;
            return;
        }
        actual = actual->sig;
    }
}

void podioYEstadisticas(struct NodoComp* arreglo[], int total_dis) {
    for (int i = 0; i < total_dis; i++) {
        printf("\n--- DISCIPLINA %d ---\n", i + 1);
        int inscriptos = 0, presentes = 0;
        float min_puntaje = 9999.0;
        
        // Variables rudimentarias para Top 3 (Asumiendo puntajes no negativos)
        float max1 = -1, max2 = -1, max3 = -1;
        int d1 = 0, d2 = 0, d3 = 0;
        
        struct NodoComp* actual = arreglo[i];
        while (actual != NULL) {
            inscriptos++;
            if (actual->dato.presente == 1) {
                presentes++;
                float p = actual->dato.puntaje;
                if (p < min_puntaje) min_puntaje = p;
                
                // Calculo de podio en una pasada
                if (p > max1) {
                    max3 = max2; d3 = d2;
                    max2 = max1; d2 = d1;
                    max1 = p; d1 = actual->dato.dni;
                } else if (p > max2) {
                    max3 = max2; d3 = d2;
                    max2 = p; d2 = actual->dato.dni;
                } else if (p > max3) {
                    max3 = p; d3 = actual->dato.dni;
                }
            }
            actual = actual->sig;
        }
        
        printf("Inscriptos: %d | Presentes: %d\n", inscriptos, presentes);
        if (presentes > 0) {
            printf("Menor puntaje obtenido: %.2f\n", min_puntaje);
            printf("PODIO:\n");
            printf("  1ro) DNI %d (%.2f pts)\n", d1, max1);
            if (max2 != -1) printf("  2do) DNI %d (%.2f pts)\n", d2, max2);

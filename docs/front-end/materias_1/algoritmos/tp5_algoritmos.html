Ejercicio 1: Búsqueda de clave carácter

Realizar un algoritmo que permita buscar en un vector de 1500 posiciones la ocurrencia de una clave de tipo carácter introducida por el usuario.

Paso a Paso:

  • Búsqueda Secuencial: Se recorre todo el vector utilizando un bucle for, ya que el vector no está garantizado que esté ordenado.
  • Por cada coincidencia, se incrementa un contador y se puede imprimir su posición. (Se usa un tamaño menor para la prueba).
#include <stdio.h>

#define MAX 10 // Simula las 1500 posiciones

int buscarCaracter(char vector[], int tam, char clave) {
    int ocurrencias = 0;
    for (int i = 0; i < tam; i++) {
        if (vector[i] == clave) {
            printf("Coincidencia encontrada en la posicion: %d\n", i);
            ocurrencias++;
        }
    }
    return ocurrencias;
}

int main() {
    char letras[MAX] = {'a', 'b', 'c', 'x', 'e', 'x', 'g', 'h', 'x', 'j'};
    char buscado;
    
    printf("Ingrese el caracter a buscar: ");
    scanf(" %c", &buscado);
    
    int encontradas = buscarCaracter(letras, MAX, buscado);
    
    if (encontradas == 0) {
        printf("El caracter '%c' no se encuentra en el vector.\n", buscado);
    } else {
        printf("Total de ocurrencias: %d\n", encontradas);
    }
    
    return 0;
}

Ejercicio 2: Selección y Burbuja Bidireccional

Implementar como procedimientos los algoritmos de ordenación por selección y burbuja, de modo que uno de ellos trabaje en sentido ascendente y el otro en sentido descendente. Utilizarlos en un programa que lea un vector de N posiciones y mostrarlo ordenado en ambos sentidos.

#include <stdio.h>

#define N 5

// Ordenamiento por Seleccion (Ascendente)
void seleccionAscendente(int vec[], int tam) {
    int i, j, min_idx, aux;
    for (i = 0; i < tam - 1; i++) {
        min_idx = i;
        for (j = i + 1; j < tam; j++) {
            if (vec[j] < vec[min_idx]) {
                min_idx = j;
            }
        }
        aux = vec[i];
        vec[i] = vec[min_idx];
        vec[min_idx] = aux;
    }
}

// Ordenamiento por Burbuja (Descendente)
void burbujaDescendente(int vec[], int tam) {
    int i, j, aux;
    for (i = 0; i < tam - 1; i++) {
        for (j = 0; j < tam - i - 1; j++) {
            if (vec[j] < vec[j + 1]) { // Cambia la condicion a menor que
                aux = vec[j];
                vec[j] = vec[j + 1];
                vec[j + 1] = aux;
            }
        }
    }
}

void imprimirVector(int vec[], int tam) {
    for (int i = 0; i < tam; i++) {
        printf("%d ", vec[i]);
    }
    printf("\n");
}

int main() {
    int numeros[N] = {34, 12, 89, 5, 23};
    int copia[N];
    
    // Copiamos para no perder el original
    for (int i = 0; i < N; i++) copia[i] = numeros[i];
    
    printf("Vector original: ");
    imprimirVector(numeros, N);
    
    seleccionAscendente(numeros, N);
    printf("Ordenado Ascendente (Seleccion): ");
    imprimirVector(numeros, N);
    
    burbujaDescendente(copia, N);
    printf("Ordenado Descendente (Burbuja):  ");
    imprimirVector(copia, N);
    
    return 0;
}

Ejercicio 3: Ordenar Matriz por Columna Indice

Se desea cargar una matriz de 14x50, ordenarla por una columna indicada, y luego imprimir la matriz resultado. La ordenación también debe realizarla un procedimiento.

Paso a Paso:

  • Se ordenan las filas completas en base al valor de una columna específica.
  • Se utiliza el algoritmo de burbuja iterando sobre las filas. Si la celda de la columna elegida es mayor que la de abajo, se intercambian todos los elementos de ambas filas usando un bucle k adicional.
#include <stdio.h>

#define FILAS 4
#define COLS 3

void ordenarMatrizPorColumna(int mat[FILAS][COLS], int col_idx) {
    int i, j, k, aux;
    
    // Burbuja adaptada para mover filas enteras
    for (i = 0; i < FILAS - 1; i++) {
        for (j = 0; j < FILAS - i - 1; j++) {
            if (mat[j][col_idx] > mat[j + 1][col_idx]) {
                // Intercambiar toda la fila
                for (k = 0; k < COLS; k++) {
                    aux = mat[j][k];
                    mat[j][k] = mat[j + 1][k];
                    mat[j + 1][k] = aux;
                }
            }
        }
    }
}

void imprimirMatriz(int mat[FILAS][COLS]) {
    for (int i = 0; i < FILAS; i++) {
        for (int j = 0; j < COLS; j++) {
            printf("%4d ", mat[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int matriz[FILAS][COLS] = {
        {10, 50, 30},
        {20, 10, 80},
        {30, 90, 10},
        {40, 20, 40}
    };
    int col_elegida;
    
    printf("Matriz Original:\n");
    imprimirMatriz(matriz);
    
    printf("\nIngrese la columna por la cual ordenar (0 a %d): ", COLS - 1);
    scanf("%d", &col_elegida);
    
    if (col_elegida >= 0 && col_elegida < COLS) {
        ordenarMatrizPorColumna(matriz, col_elegida);
        printf("\nMatriz Ordenada por la columna %d:\n", col_elegida);
        imprimirMatriz(matriz);
    } else {
        printf("Columna invalida.\n");
    }
    
    return 0;
}

Ejercicio 5: Búsqueda Recursiva del Precio de un Libro

Dado un vector de libros (titulo, autor, código, precio) retornar el precio de un libro determinado (búsqueda secuencial en vector pero recursiva).

Paso a Paso:

  • La recursividad funciona avanzando el índice idx en cada llamada.
  • Caso base 1: Si idx supera el tamaño del arreglo, no se encontró.
  • Caso base 2: Si el código coincide, retorna el precio.
  • Caso recursivo: Llama a la función sumando 1 al índice.
#include <stdio.h>

struct Libro {
    char titulo[30];
    char autor[30];
    int codigo;
    float precio;
};

// Funcion recursiva
float buscarPrecioRecursivo(struct Libro inv[], int tam, int idx, int cod_buscado) {
    // Caso base 1: Se llego al final del arreglo sin encontrarlo
    if (idx >= tam) {
        return -1.0; 
    }
    // Caso base 2: Se encontro el libro
    if (inv[idx].codigo == cod_buscado) {
        return inv[idx].precio;
    }
    // Llamada recursiva avanzando al siguiente indice
    return buscarPrecioRecursivo(inv, tam, idx + 1, cod_buscado);
}

int main() {
    struct Libro biblioteca[3] = {
        {"Cien anos", "Garcia Marquez", 101, 2500.50},
        {"El Aleph", "Borges", 102, 1800.00},
        {"Rayuela", "Cortazar", 103, 3100.75}
    };
    
    int cod;
    printf("Ingrese codigo del libro a buscar: ");
    scanf("%d", &cod);
    
    // Inicia la busqueda recursiva desde el indice 0
    float precio = buscarPrecioRecursivo(biblioteca, 3, 0, cod);
    
    if (precio != -1.0) {
        printf("El precio del libro es: $%.2f\n", precio);
    } else {
        printf("Libro no encontrado.\n");
    }
    
    return 0;
}

Ejercicio 6: Matriz de Palabras (Orden Alfabético)

En una matriz de 10x20 se cargan 20 palabras. Diseñar un procedimiento que ordene las filas de la matriz de modo que las palabras cargadas puedan leerse en orden alfabético de arriba hacia abajo.

Una matriz de caracteres de dos dimensiones en C equivale a un arreglo de strings. Se utiliza strcmp para comparar alfabéticamente y strcpy para intercambiar las palabras.

#include <stdio.h>
#include <string.h>

#define FILAS 5 // Reducido para la demostracion
#define MAX_LETRAS 20

void ordenarPalabras(char matriz[FILAS][MAX_LETRAS]) {
    int i, j;
    char aux[MAX_LETRAS];
    
    // Algoritmo de Burbuja aplicado a cadenas de texto
    for (i = 0; i < FILAS - 1; i++) {
        for (j = 0; j < FILAS - i - 1; j++) {
            // strcmp devuelve > 0 si la primera cadena es mayor alfabeticamente
            if (strcmp(matriz[j], matriz[j + 1]) > 0) {
                // Intercambio completo de la fila (palabra)
                strcpy(aux, matriz[j]);
                strcpy(matriz[j], matriz[j + 1]);
                strcpy(matriz[j + 1], aux);
            }
        }
    }
}

int main() {
    char palabras[FILAS][MAX_LETRAS] = {
        "Zapato",
        "Arbol",
        "Mesa",
        "Computadora",
        "Silla"
    };
    
    printf("Palabras Desordenadas:\n");
    for (int i = 0; i < FILAS; i++) {
        printf("%s\n", palabras[i]);
    }
    
    ordenarPalabras(palabras);
    
    printf("\nPalabras Ordenadas Alfabeticamente:\n");
    for (int i = 0; i < FILAS; i++) {
        printf("%s\n", palabras[i]);
    }
    
    return 0;
}

Ejercicio 7: Matriz de Números como Caracteres (Alineación Derecha)

Rehacer el ejercicio anterior con una matriz de números. En cada elemento se carga un solo dígito pero la alineación es hacia la derecha (relleno con 0). Comparar el resultado con el orden usual de números.

Al alinear con ceros a la izquierda (ej. `00045` vs `00120`), la comparación lexicográfica de texto (usando strcmp) coincide perfectamente con la comparación numérica real. Esta es la razón de rellenar con ceros a la izquierda en formatos de texto.

#include <stdio.h>
#include <string.h>

#define FILAS 4
#define MAX_DIGITOS 6

void ordenarNumerosTexto(char matriz[FILAS][MAX_DIGITOS]) {
    int i, j;
    char aux[MAX_DIGITOS];
    
    for (i = 0; i < FILAS - 1; i++) {
        for (j = 0; j < FILAS - i - 1; j++) {
            if (strcmp(matriz[j], matriz[j + 1]) > 0) {
                strcpy(aux, matriz[j]);
                strcpy(matriz[j], matriz[j + 1]);
                strcpy(matriz[j + 1], aux);
            }
        }
    }
}

int main() {
    // Numeros rellenados con 0 a la izquierda (alineacion derecha logica)
    char numeros_txt[FILAS][MAX_DIGITOS] = {
        "00045",
        "00120",
        "00008",
        "00099"
    };
    
    printf("Matriz de Numeros Original:\n");
    for (int i = 0; i < FILAS; i++) {
        printf("%s\n", numeros_txt[i]);
    }
    
    ordenarNumerosTexto(numeros_txt);
    
    printf("\nMatriz Ordenada (Lexicograficamente):\n");
    for (int i = 0; i < FILAS; i++) {
        printf("%s\n", numeros_txt[i]);
    }
    
    printf("\nEl orden alfabetico coincide con el orden numerico gracias al relleno de ceros.\n");
    
    return 0;
}

Ejercicio 8: Menú Interactivo de Ordenación

Realizar un programa que permita al usuario cargar 35 números enteros en un vector y decidir mediante un menú qué método de ordenación utilizar y si el orden debe ser ascendente o descendente.

#include <stdio.h>

#define TAM 5 // Reducido para prueba

void burbuja(int v[], int n, int ascendente) {
    int aux;
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            int condicion = ascendente ? (v[j] > v[j+1]) : (v[j] < v[j+1]);
            if(condicion) {
                aux = v[j];
                v[j] = v[j+1];
                v[j+1] = aux;
            }
        }
    }
}

void seleccion(int v[], int n, int ascendente) {
    int aux, min_max_idx;
    for(int i = 0; i < n - 1; i++) {
        min_max_idx = i;
        for(int j = i + 1; j < n; j++) {
            int condicion = ascendente ? (v[j] < v[min_max_idx]) : (v[j] > v[min_max_idx]);
            if(condicion) min_max_idx = j;
        }
        aux = v[i];
        v[i] = v[min_max_idx];
        v[min_max_idx] = aux;
    }
}

void imprimir(int v[], int n) {
    for(int i = 0; i < n; i++) printf("%d ", v[i]);
    printf("\n");
}

int main() {
    int arr[TAM] = {8, 2, 9, 1, 5};
    int original[TAM] = {8, 2, 9, 1, 5};
    int opcion_metodo, opcion_orden;
    
    do {
        // Restaurar vector original para cada prueba
        for(int i=0; i<TAM; i++) arr[i] = original[i];
        
        printf("\nVector base: "); imprimir(arr, TAM);
        printf("1. Burbuja\n2. Seleccion\n3. Salir\nOpcion: ");
        scanf("%d", &opcion_metodo);
        
        if(opcion_metodo == 1 || opcion_metodo == 2) {
            printf("Orden (1: Ascendente, 0: Descendente): ");
            scanf("%d", &opcion_orden);
            
            if(opcion_metodo == 1) burbuja(arr, TAM, opcion_orden);
            if(opcion_metodo == 2) seleccion(arr, TAM, opcion_orden);
            
            printf("Vector Ordenado: ");
            imprimir(arr, TAM);
        }
    } while(opcion_metodo != 3);

    return 0;
}

Ejercicio 10: Búsqueda Inteligente (Secuencial vs Dicotómica)

Codificar un programa que permita cargar, ordenar y buscar sobre un vector de reales. La búsqueda debe ser inteligente: si el vector no está ordenado, usa búsqueda lineal; si está ordenado, usa búsqueda dicotómica (binaria).

Se maneja una bandera esta_ordenado para saber qué algoritmo aplicar. Al insertar un número nuevo (fuera de orden), el arreglo perdería su garantía de estar ordenado.

#include <stdio.h>

#define MAX 10

void ordenar(float v[], int n) {
    float aux;
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(v[j] > v[j+1]) {
                aux = v[j];
                v[j] = v[j+1];
                v[j+1] = aux;
            }
        }
    }
}

int busquedaSecuencial(float v[], int n, float buscado) {
    printf("[Usando Busqueda Secuencial Lineal]... ");
    for(int i = 0; i < n; i++) {
        if(v[i] == buscado) return i;
    }
    return -1;
}

int busquedaDicotomica(float v[], int n, float buscado) {
    printf("[Usando Busqueda Dicotomica / Binaria]... ");
    int inicio = 0, fin = n - 1, medio;
    while(inicio <= fin) {
        medio = (inicio + fin) / 2;
        if(v[medio] == buscado) return medio;
        if(v[medio] < buscado) inicio = medio + 1;
        else fin = medio - 1;
    }
    return -1;
}

int main() {
    float arreglo[MAX] = {10.5, 3.2, 8.1, 1.0, 5.5};
    int cantidad = 5;
    int esta_ordenado = 0; // Bandera de estado
    int opcion;
    float buscar;

    do {
        printf("\n1. Ver Vector\n2. Ordenar Vector\n3. Buscar numero\n4. Salir\nOpcion: ");
        scanf("%d", &opcion);
        
        if (opcion == 1) {
            for(int i=0; i<cantidad; i++) printf("%.2f ", arreglo[i]);
            printf("\n");
        } 
        else if (opcion == 2) {
            ordenar(arreglo, cantidad);
            esta_ordenado = 1; // Actualizamos estado
            printf("Vector ordenado exitosamente.\n");
        } 
        else if (opcion == 3) {
            printf("Ingrese numero real a buscar: ");
            scanf("%f", &buscar);
            
            int pos;
            if (esta_ordenado) {
                pos = busquedaDicotomica(arreglo, cantidad, buscar);
            } else {
                pos = busquedaSecuencial(arreglo, cantidad, buscar);
            }
            
            if(pos != -1) printf("Encontrado en el indice %d\n", pos);
            else printf("No encontrado.\n");
        }
    } while (opcion != 4);

    return 0;
}

Ejercicio 1: Gestión de Hotel

Un hotel maneja 100 habitaciones con `Nro, piso, tipo, precioXdia, confort, estado`. Realizar: a) Buscar disponibles, b) Imprimir estados ordenadas por tipo, c) Liquidar habitación (ingresando días ocupada).

#include <stdio.h>

#define MAX_HAB 100

// Estructura local sin variables globales
struct Habitacion {
    int nro;
    int piso;
    int tipo;      // 1: Simple, 2: Doble
    float precioXdia;
    int estado;    // 0: Disponible, 1: Ocupada
};

void buscarDisponibles(struct Habitacion hotel[], int n) {
    printf("\n--- Habitaciones Disponibles ---\n");
    for(int i = 0; i < n; i++) {
        if(hotel[i].estado == 0 && hotel[i].nro != 0) {
            printf("Hab %d | Piso %d | Tipo %d | $%.2f\n", 
                   hotel[i].nro, hotel[i].piso, hotel[i].tipo, hotel[i].precioXdia);
        }
    }
}

void liquidarHabitacion(struct Habitacion hotel[], int n, int num_hab, int dias) {
    for(int i = 0; i < n; i++) {
        if(hotel[i].nro == num_hab) {
            if(hotel[i].estado == 1) {

Ejercicio 2: Transacciones Bancarias (Tarjeta de Crédito)

Arreglo con 5000 transacciones: `Nro Cuenta, Fecha, Monto, Comerciante`. Se pide: a) Ordenar por Nro Cuenta, Fecha y Comerciante, b) Cantidad de clientes, c) Listar totales de monto por cliente.

Paso a Paso:

  • Se utiliza el concepto de Corte de Control, aprovechando que el arreglo ya se encuentra ordenado por el criterio principal (Número de Cuenta).
  • Esto permite agrupar y acumular montos de un mismo cliente en un único bucle, detectando cuando el número de cuenta cambia para imprimir el total acumulado y reiniciar el contador para el siguiente cliente.
#include <stdio.h>

#define MAX_TRANS 6 

struct Transaccion {
    int nro_cuenta;
    int fecha;      // Formato YYYYMMDD
    float monto;
    int comerciante;
};

// Burbuja con multiples criterios
void ordenarTransacciones(struct Transaccion tx[], int n) {
    struct Transaccion aux;
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (tx[j].nro_cuenta > tx[j+1].nro_cuenta ||
               (tx[j].nro_cuenta == tx[j+1].nro_cuenta && tx[j].fecha > tx[j+1].fecha) ||
               (tx[j].nro_cuenta == tx[j+1].nro_cuenta && tx[j].fecha == tx[j+1].fecha && tx[j].comerciante > tx[j+1].comerciante)) {
                aux = tx[j];
                tx[j] = tx[j+1];
                tx[j+1] = aux;
            }
        }
    }
}

// Corte de control para calcular totales
void reporteClientes(struct Transaccion tx[], int n) {
    if(n == 0) return;
    
    int cuenta_actual = tx[0].nro_cuenta;
    float total_acumulado = 0.0;
    int clientes_distintos = 1;

    printf("\n--- TOTALES POR CLIENTE ---\n");
    for(int i = 0; i < n; i++) {
        if(tx[i].nro_cuenta == cuenta_actual) {
            total_acumulado += tx[i].monto;
        } else {
            // Cambio de cuenta (corte): se imprime total del anterior
            printf("Cuenta: %d | Total Gastos: $%.2f\n", cuenta_actual, total_acumulado);
            
            // Se resetean acumuladores para la nueva cuenta
            cuenta_actual = tx[i].nro_cuenta;
            total_acumulado = tx[i].monto;
            clientes_distintos++;
        }
    }
    // Impresion de la ultima cuenta que quedo pendiente en el acumulador
    printf("Cuenta: %d | Total Gastos: $%.2f\n", cuenta_actual, total_acumulado);
    printf("\nCantidad total de clientes (Cuentas activas): %d\n", clientes_distintos);
}

int main() {
    struct Transaccion historial[MAX_TRANS] = {
        {1001, 20261012, 1500.50, 99},
        {1002, 20261011, 2000.00, 15},
        {1001, 20261010, 500.00,  99},
        {1003, 20261012, 1000.00, 10},
        {1001, 20261012, 100.00,  50},
        {1003, 20261014, 5000.00, 10}
    };
    
    ordenarTransacciones(historial, MAX_TRANS);
    reporteClientes(historial, MAX_TRANS);
    
    return 0;
}

Ejercicio 3: Inventario Base de Datos (Mantenimiento Ordenado)

Simular base de datos de productos con un arreglo. Mantenerlos SIEMPRE ordenados por Nro de Ítem. Permitir: a) Insertar, b) Borrar, c) Modificar, d) Buscar (Dicotómica).

La clave de este ejercicio es la función de Inserción, la cual no debe agregar el elemento al final y reordenar todo desde cero (ineficiente), sino aplicar una lógica de desplazamiento tipo Insertion Sort: desplazar los elementos mayores hacia la derecha para hacer un "hueco" e insertar el nuevo elemento directamente en su posición ordenada.

#include <stdio.h>

#define MAX_ITEMS 10000

struct Producto {
    int nro_item;
    float costo;
    int stock_act;
};

// Busqueda Binaria eficiente (el arreglo siempre esta ordenado)
int buscarProducto(struct Producto inv[], int cantidad, int id_buscado) {
    int inicio = 0, fin = cantidad - 1, medio;
    while(inicio <= fin) {
        medio = (inicio + fin) / 2;
        if(inv[medio].nro_item == id_buscado) {
            return medio; // Retorna el indice exacto
        } else if(inv[medio].nro_item < id_buscado) {
            inicio = medio + 1;
        } else {
            fin = medio - 1;
        }
    }
    return -1; // No existe
}

// Insercion con desplazamiento (Insertion logic)
void insertarProducto(struct Producto inv[], int *cantidad, struct Producto nuevo) {
    if(*cantidad >= MAX_ITEMS) {
        printf("Error: Base de datos llena.\n");
        return;
    }
    
    // Si ya existe, no se inserta duplicado
    if(buscarProducto(inv, *cantidad, nuevo.nro_item) != -1) {
        printf("Error: El producto %d ya existe.\n", nuevo.nro_item);
        return;
    }
    
    int i = *cantidad - 1;
    // Desplazar elementos mayores hacia la derecha
    while(i >= 0 && inv[i].nro_item > nuevo.nro_item) {
        inv[i + 1] = inv[i];
        i--;
    }
    // Insercion en la posicion correcta
    inv[i + 1] = nuevo;
    (*cantidad)++;
    printf("Producto %d insertado con exito.\n", nuevo.nro_item);
}

void borrarProducto(struct Producto inv[], int *cantidad, int id_borrar) {
    int pos = buscarProducto(inv, *cantidad, id_borrar);
    if(pos == -1) {
        printf("Error: Producto %d no existe.\n", id_borrar);
        return;
    }
    
    // Desplazar elementos hacia la izquierda pisando el borrado
    for(int i = pos; i < *cantidad - 1; i++) {
        inv[i] = inv[i + 1];
    }
    (*cantidad)--;
    printf("Producto %d eliminado con exito.\n", id_borrar);
}

int main() {
    struct Producto inventario[MAX_ITEMS];
    int cantidad = 0;
    
    struct Producto p1 = {105, 50.0, 100};
    struct Producto p2 = {102, 25.5, 50};
    struct Producto p3 = {108, 90.0, 10};
    
    printf("--- INSERCIONES ---\n");
    insertarProducto(inventario, &cantidad, p1);
    insertarProducto(inventario, &cantidad, p2); 
    insertarProducto(inventario, &cantidad, p3);
    
    printf("\n--- ESTADO DEL INVENTARIO (Siempre Ordenado) ---\n");
    for(int i = 0; i < cantidad; i++) {
        printf("Item: %d | Costo: $%.2f\n", inventario[i].nro_item, inventario[i].costo);
    }
    
    printf("\n--- BORRADO ---\n");
    borrarProducto(inventario, &cantidad, 105);
    
    printf("\n--- ESTADO DEL INVENTARIO FINAL ---\n");
    for(int i = 0; i < cantidad; i++) {
        printf("Item: %d | Costo: $%.2f\n", inventario[i].nro_item, inventario[i].costo);
    }
    
    return 0;
}

Ejercicio 4: Vector de Alumnos (Aprobados y Promedio con Recursión)

Dado un vector con los datos de los alumnos de un curso, determinar la cantidad de aprobados y calcular la nota promedio del curso usando recursión de forma estricta (sin bucles iterativos).

Se utilizan dos funciones recursivas independientes que van avanzando un índice idx y acumulando sus resultados hasta que el índice alcanza el tamaño total del arreglo (caso base).

#include <stdio.h>

struct Alumno {
    char nombre[30];
    float nota;
};

// Funcion recursiva para contar aprobados (nota >= 6)
int contarAprobados(struct Alumno clase[], int n, int idx) {
    if (idx >= n) return 0; // Caso Base
    
    int es_aprobado = (clase[idx].nota >= 6.0) ? 1 : 0;
    return es_aprobado + contarAprobados(clase, n, idx + 1);
}

// Funcion recursiva para sumar notas (para el promedio)
float sumarNotas(struct Alumno clase[], int n, int idx) {
    if (idx >= n) return 0.0; // Caso Base
    
    return clase[idx].nota + sumarNotas(clase, n, idx + 1);
}

int main() {
    struct Alumno curso[4] = {
        {"Perez", 7.5},
        {"Gomez", 4.0},
        {"Lopez", 9.0},
        {"Diaz", 5.5}
    };
    int total_alumnos = 4;
    
    int aprobados = contarAprobados(curso, total_alumnos, 0);
    float suma_total = sumarNotas(curso, total_alumnos, 0);
    float promedio = suma_total / total_alumnos;
    
    printf("Total de alumnos aprobados: %d\n", aprobados);
    printf("Nota promedio del curso: %.2f\n", promedio);

    return 0;
}

Ejercicio 5: Búsqueda Recursiva de Stock Nulo y Marcas

Dado un vector de productos (cod, marca, nombre, stock) retornar la cantidad de productos de una marca dada e imprimir el nombre de todos los que tienen stock nulo. Usar recursión para la solución.

#include <stdio.h>
#include <string.h>

struct Producto {
    int cod;
    char marca[30];
    char nombre[30];
    int stock;
};

// Funcion recursiva que hace ambas tareas solicitadas en la misma pasada
int procesarProductosRecursivo(struct Producto inv[], int n, int idx, char marca_buscada[]) {
    if (idx >= n) return 0; // Caso Base: fin del arreglo
    
    // Accion 1: Imprimir si el stock es nulo
    if (inv[idx].stock == 0) {
        printf(" -> ALERTA: El producto '%s' (Cod: %d) tiene STOCK NULO.\n", 
               inv[idx].nombre, inv[idx].cod);
    }
    
    // Accion 2: Contar si pertenece a la marca
    int pertenece_marca = 0;
    if (strcmp(inv[idx].marca, marca_buscada) == 0) {
        pertenece_marca = 1;
    }
    
    // Llamada recursiva avanzando al siguiente
    return pertenece_marca + procesarProductosRecursivo(inv, n, idx + 1, marca_buscada);
}

    int cant_marca = procesarProductosRecursivo(inventario, 5, 0, marca_obj);

Ejercicio 6: Control de Tránsito

20 inspectores con hasta 2000 actas cada uno. Registros: `nroActa, dominio, tipoInfraccion, idInspector`. Descargar a un soporte único, ordenar por tipoInfraccion e inspector, y mostrar subtotales y porcentajes (Corte de Control).

#include <stdio.h>

#define TOTAL_ACTAS 6 // Reducido para ejemplo

struct Acta {
    int nroActa;
    int tipoInfraccion;
    int idInspector;
};

// Ordenar por tipoInfraccion (1er criterio) y luego idInspector (2do criterio)
void ordenarActas(struct Acta actas[], int n) {
    struct Acta aux;
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(actas[j].tipoInfraccion > actas[j+1].tipoInfraccion || 
              (actas[j].tipoInfraccion == actas[j+1].tipoInfraccion && actas[j].idInspector > actas[j+1].idInspector)) {
                aux = actas[j];
                actas[j] = actas[j+1];
                actas[j+1] = aux;
            }
        }
    }
}

// Logica de Corte de Control
void reporteInfracciones(struct Acta actas[], int n) {
    if (n == 0) return;
    
    int i = 0;
    int total_general = 0;
    
    while(i < n) {
        int tipo_actual = actas[i].tipoInfraccion;
        int total_tipo = 0;
        
        printf("\nInfraccion tipo %d:\n", tipo_actual);
        
        while(i < n && actas[i].tipoInfraccion == tipo_actual) {
            int inspector_actual = actas[i].idInspector;
            int actas_inspector = 0;
            
            while(i < n && actas[i].tipoInfraccion == tipo_actual && actas[i].idInspector == inspector_actual) {

                total_general++;

            }

            printf("  Inspector %d: %d actas\n", inspector_actual, actas_inspector);
        }
        printf("Total infracciones tipo %d: %d\n", tipo_actual, total_tipo);
    }
    printf("\nCantidad total de infracciones general: %d\n", total_general);
}

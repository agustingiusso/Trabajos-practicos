Ejercicio 1: Fábrica de Llantas

En una fábrica de llantas se confeccionan entre 500 y 1000 llantas, por cada uno de los 4 turnos de 6 horas cada uno, con un tamaño de 14 a 19. Las buenas se apilan por tamaño, y las rayadas (con fallas) en otra pila. Cada 10 llantas se arman cajas. Informar por turno: cajas armadas por tamaño (con fallas y sin fallas), y el porcentaje del remanente no embalado.

#include <stdio.h>

// Uso de constantes locales mediante define para evitar variables globales
#define TURNOS 4
#define TAMANOS 6 // Tamaños del 14 al 19

void procesarLlantas(int sin_falla[TURNOS][TAMANOS], int con_falla[TURNOS][TAMANOS]) {
    int cajas_sin[TURNOS][TAMANOS] = {0};
    int cajas_con[TURNOS][TAMANOS] = {0};
    int remanente_sin[TAMANOS] = {0};
    int prod_diaria[TAMANOS] = {0};
    
    // Simulación del empaquetado (cajas de 10)
    for(int t = 0; t < TURNOS; t++) {
        for(int tam = 0; tam < TAMANOS; tam++) {
            cajas_sin[t][tam] = sin_falla[t][tam] / 10;
            remanente_sin[tam] += sin_falla[t][tam] % 10; // Acumular remanente
            
            cajas_con[t][tam] = con_falla[t][tam] / 10;
            
            prod_diaria[tam] += sin_falla[t][tam] + con_falla[t][tam];
        }
    }
    
    printf("\n--- REPORTE DE CAJAS POR TURNO ---\n");
    for(int t = 0; t < TURNOS; t++) {
        printf("Turno %d:\n", t + 1);
        for(int tam = 0; tam < TAMANOS; tam++) {
            printf("  Tamano %d -> Cajas Buenas: %d | Cajas con Falla: %d\n", 
                   tam + 14, cajas_sin[t][tam], cajas_con[t][tam]);
        }
    }
    
    printf("\n--- PORCENTAJE DE REMANENTE SIN FALLA ---\n");
    for(int tam = 0; tam < TAMANOS; tam++) {
        if(prod_diaria[tam] > 0) {
            float porc = ((float)remanente_sin[tam] / prod_diaria[tam]) * 100;
            printf("  Tamano %d: %.2f%%\n", tam + 14, porc);
        }
    }
}

int main() {
    // Variables locales para almacenar produccion
    int llantas_buenas[TURNOS][TAMANOS] = {
        {120, 105, 200, 150, 90, 80},
        {110, 95, 180, 140, 85, 75},
        {130, 115, 210, 160, 95, 85},
        {100, 85, 190, 130, 70, 60}
    };
    int llantas_malas[TURNOS][TAMANOS] = {
        {12, 10, 20, 15, 9, 8},
        {11, 9, 18, 14, 8, 7},
        {13, 11, 21, 16, 9, 8},
        {10, 8, 19, 13, 7, 6}
    };
    
    procesarLlantas(llantas_buenas, llantas_malas);
    return 0;
}

Ejercicio 2: Salón de Fiestas (Postres)

En un salón hay 15 mesas de 8 sillas. Cada consola permite pedidos de postre preferido o alternativo. Al 3er pedido del mismo postre se indica 'Glotón'. Informar por cada mesa: cantidad de invitados que comieron postre preferido, postre alternativo y cantidad de 'Glotones'.

#include <stdio.h>

#define MESAS 15

void resumenConsumo(int preferido[], int alternativo[], int glotones[]) {
    printf("\n--- REPORTE DE POSTRES POR MESA ---\n");
    for(int i = 0; i < MESAS; i++) {
        printf("Mesa %d:\n", i + 1);
        printf("  Comieron Preferido: %d invitados\n", preferido[i]);
        printf("  Comieron Alternativo: %d invitados\n", alternativo[i]);
        printf("  Identificados como Glotones: %d\n", glotones[i]);
    }
}

int main() {
    // Arrays locales para contabilizar estadisticas por mesa
    int cant_preferido[MESAS] = {0};
    int cant_alternativo[MESAS] = {0};
    int cant_glotones[MESAS] = {0};
    int num_mesa, tipo_pedido, es_gloton;

    printf("Simulacion de atencion (Ingrese Mesa 0 para terminar):\n");
    do {
        printf("Mesa (1-15): ");
        scanf("%d", &num_mesa);
        
        if(num_mesa >= 1 && num_mesa <= 15) {
            printf("Pedido (1: Preferido, 2: Alternativo): ");
            scanf("%d", &tipo_pedido);
            printf("Es su tercer pedido del mismo postre? (1: Si, 0: No): ");

Ejercicio 3: Operaciones Básicas sobre un Vector

Implementar programas adecuados para realizar las siguientes operaciones sobre un vector de enteros: a) Buscar el máximo valor, b) Determinar si un valor dado existe, c) Determinar el elemento central (o el menor de los centrales si es par).

#include <stdio.h>

// a) Buscar el máximo
int buscarMaximo(int vec[], int tam) {
    int max = vec[0];
    for (int i = 1; i < tam; i++) {
        if (vec[i] > max) {
            max = vec[i];
        }
    }
    return max;
}

// b) Buscar si existe un valor
int existeValor(int vec[], int tam, int valor) {
    for (int i = 0; i < tam; i++) {
        if (vec[i] == valor) return 1; // Verdadero
    }
    return 0; // Falso
}

// c) Obtener el elemento central
int elementoCentral(int vec[], int tam) {
    if (tam % 2 != 0) {
        return vec[tam / 2];
    } else {
        int c1 = vec[(tam / 2) - 1];
        int c2 = vec[tam / 2];
        return (c1 < c2) ? c1 : c2;
    }
}

int main() {
    int numeros[] = {12, 45, 7, 89, 23, 56};
    int tamano = 6;
    
    printf("Vector: [12, 45, 7, 89, 23, 56]\n");
    printf("Maximo: %d\n", buscarMaximo(numeros, tamano));
    
    int buscar = 7;
    if (existeValor(numeros, tamano, buscar)) {
        printf("El valor %d SI existe en el vector.\n", buscar);
    }
    
    printf("Elemento central (o menor de centrales): %d\n", elementoCentral(numeros, tamano));
    
    return 0;
}

Ejercicio 5: Control de Restaurante (50 mesas)

Un restaurante lleva el control de 50 mesas. Se cobra cubierto por cada comensal y se agrega el monto de los pedidos. Al retirarse se cobra el acumulado. Considerar la falta de mesas libres.

#include <stdio.h>

#define MESAS 50
#define PRECIO_CUBIERTO 150.0

int main() {
    float cuentas[MESAS] = {0.0};
    int ocupadas[MESAS] = {0}; // 0 = Libre, 1 = Ocupada
    int opcion, num_mesa, comensales;
    float monto;

    do {
        printf("\n1. Asignar Mesa\n2. Agregar Pedido\n3. Liberar y Cobrar\n4. Salir\nOpcion: ");
        scanf("%d", &opcion);

        switch (opcion) {
            case 1:
                // Buscar mesa libre
                num_mesa = -1;
                for (int i = 0; i < MESAS; i++) {
                    if (ocupadas[i] == 0) {
                        num_mesa = i;
                        break;
                    }
                }
                if (num_mesa != -1) {
                    printf("Mesa %d asignada. Cuantos comensales?: ", num_mesa + 1);
                    scanf("%d", &comensales);
                    ocupadas[num_mesa] = 1;
                    cuentas[num_mesa] = comensales * PRECIO_CUBIERTO;
                } else {
                    printf("No hay mesas libres.\n");
                }
                break;
            case 2:
                printf("Numero de mesa (1-50): ");
                scanf("%d", &num_mesa);
                if (num_mesa >= 1 && num_mesa <= MESAS && ocupadas[num_mesa - 1]) {
                    printf("Monto del pedido: $");
                    scanf("%f", &monto);
                    cuentas[num_mesa - 1] += monto;
                    printf("Pedido agregado.\n");
                } else {
                    printf("Mesa invalida o libre.\n");
                }
                break;
            case 3:
                printf("Numero de mesa a cobrar (1-50): ");
                scanf("%d", &num_mesa);
                if (num_mesa >= 1 && num_mesa <= MESAS && ocupadas[num_mesa - 1]) {
                    printf("Total a cobrar en Mesa %d: $%.2f\n", num_mesa, cuentas[num_mesa - 1]);
                    ocupadas[num_mesa - 1] = 0;
                    cuentas[num_mesa - 1] = 0.0;
                } else {
                    printf("Mesa invalida o ya esta libre.\n");
                }
                break;
        }
    } while (opcion != 4);

    return 0;
}

Ejercicio 6: Operaciones Básicas sobre una Matriz

Implementar: a) Buscar el mínimo, b) Buscar si un valor existe, c) Posición de un valor (0 si no está), d) Imprimir de izq-der, arr-aba, e) Imprimir arr-aba, izq-der.

#include <stdio.h>

#define FILAS 3
#define COLS 3

void buscarMinimo(int mat[FILAS][COLS]) {
    int min = mat[0][0];
    for(int i = 0; i < FILAS; i++) {
        for(int j = 0; j < COLS; j++) {
            if(mat[i][j] < min) min = mat[i][j];
        }
    }
    printf("El valor minimo es: %d\n", min);
}

void buscarValor(int mat[FILAS][COLS], int valor) {
    int encontrado = 0;
    for(int i = 0; i < FILAS; i++) {
        for(int j = 0; j < COLS; j++) {
            if(mat[i][j] == valor) {
                printf("Valor %d encontrado en Fila %d, Col %d\n", valor, i+1, j+1);
                encontrado = 1;
            }
        }
    }
    if(!encontrado) printf("0 (El valor no esta en la matriz)\n");
}

void imprimirNormal(int mat[FILAS][COLS]) {
    printf("Recorrido Fila por Fila (Izq-Der, Arr-Aba):\n");
    for(int i = 0; i < FILAS; i++) {
        for(int j = 0; j < COLS; j++) {
            printf("%d ", mat[i][j]);
        }
        printf("\n");
    }
}

void imprimirColumnas(int mat[FILAS][COLS]) {
    printf("Recorrido Columna por Columna (Arr-Aba, Izq-Der):\n");
    for(int j = 0; j < COLS; j++) {
        for(int i = 0; i < FILAS; i++) {
            printf("%d ", mat[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int matriz[FILAS][COLS] = {
        {5, 8, 2},
        {1, 9, 3},
        {4, 7, 6}
    };
    
    imprimirNormal(matriz);
    imprimirColumnas(matriz);
    buscarMinimo(matriz);
    buscarValor(matriz, 9);
    buscarValor(matriz, 99); // No existe
    
    return 0;
}

Ejercicio 8: Operaciones con 2 Matrices Reales

Se cargan 2 matrices reales. a) Vector con suma de cada columna de la 1ra, b) Vector con promedio de cada fila de la 2da, c) Calcular el promedio de los valores de ambas matrices.

#include <stdio.h>

#define F 2
#define C 3

int main() {
    float matA[F][C] = {{1.5, 2.0, 3.5}, {0.5, 1.0, 2.0}};
    float matB[F][C] = {{4.0, 2.0, 6.0}, {1.0, 1.0, 1.0}};
    
    float suma_cols[C] = {0};
    float prom_filas[F] = {0};
    float suma_total = 0;

    // A) Suma de columnas de MatA
    for(int j = 0; j < C; j++) {
        for(int i = 0; i < F; i++) {
            suma_cols[j] += matA[i][j];
            suma_total += matA[i][j]; // Aprovechamos para la suma total
        }
    }

    // B) Promedio de filas de MatB
    for(int i = 0; i < F; i++) {
        float suma_fila = 0;
        for(int j = 0; j < C; j++) {
            suma_fila += matB[i][j];
            suma_total += matB[i][j]; // Acumulamos total
        }
        prom_filas[i] = suma_fila / C;
    }

    // C) Promedio global
    float prom_total = suma_total / (F * C * 2);

    printf("Suma por Columnas (Matriz 1): ");
    for(int j = 0; j < C; j++) printf("%.2f  ", suma_cols[j]);
    
    printf("\nPromedio por Filas (Matriz 2): ");
    for(int i = 0; i < F; i++) printf("%.2f  ", prom_filas[i]);

    printf("\nPromedio global de ambas matrices: %.2f\n", prom_total);

    return 0;
}

Ejercicio 2: Transacciones Bancarias (Tarjeta de Crédito)

Arreglo con 5000 transacciones: `Nro Cuenta, Fecha, Monto, Comerciante`. Se pide ordenar por Cuenta, Fecha y Comerciante, contar clientes, y listar totales por cliente.

#include <stdio.h>

#define MAX_TRANS 5 // Reducido para prueba local

struct Transaccion {
    int nro_cuenta;
    int fecha;      // YYYYMMDD para orden simplificado
    float monto;
    int comerciante;
};

// Algoritmo de Burbuja para el ordenamiento compuesto
void ordenarTransacciones(struct Transaccion tx[], int n) {
    struct Transaccion aux;
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            // Prioridad 1: Cuenta, Prioridad 2: Fecha, Prioridad 3: Comerciante
            if (tx[j].nro_cuenta > tx[j+1].nro_cuenta ||
               (tx[j].nro_cuenta == tx[j+1].nro_cuenta && tx[j].fecha > tx[j+1].fecha) ||
               (tx[j].nro_cuenta == tx[j+1].nro_cuenta && tx[j].fecha == tx[j+1].fecha && tx[j].comerciante > tx[j+1].comerciante)) {
                aux = tx[j];
                tx[j] = tx[j+1];
                tx[j+1] = aux;
            }
        }
    }
}

void listarTotales(struct Transaccion tx[], int n) {
    if(n == 0) return;
    
    int cuenta_actual = tx[0].nro_cuenta;
    float total_acumulado = 0.0;
    int clientes_distintos = 1;

    printf("Listado de Totales por Cuenta:\n");
    for(int i = 0; i < n; i++) {
        if(tx[i].nro_cuenta == cuenta_actual) {
            total_acumulado += tx[i].monto;
        } else {
            printf("Cuenta: %d - Total Transacciones: $%.2f\n", cuenta_actual, total_acumulado);
            cuenta_actual = tx[i].nro_cuenta;
            total_acumulado = tx[i].monto;
            clientes_distintos++;
        }
    }
    printf("Cuenta: %d - Total Transacciones: $%.2f\n", cuenta_actual, total_acumulado);
    printf("Total de Clientes (Cuentas) distintos: %d\n", clientes_distintos);
}

int main() {
    struct Transaccion historial[MAX_TRANS] = {
        {123, 20261012, 150.5, 99},
        {456, 20261011, 200.0, 15},
        {123, 20261010, 50.0, 99},
        {789, 20261012, 1000.0, 10},
        {123, 20261012, 10.0, 50}
    };
    
    ordenarTransacciones(historial, MAX_TRANS);
    listarTotales(historial, MAX_TRANS);
    
    return 0;
}

Ejercicio 3: Inventario de Comercio

Simular base de datos de productos (8000 ítems) con un arreglo de 10000. Mantenerlos ordenados por Nro de Ítem. Permitir: a) Insertar, b) Borrar, c) Modificar, d) Buscar.

#include <stdio.h>

#define MAX_ITEMS 10000

struct Producto {
    int nro_item;
    float costo;
    int stock_act;
};

// Búsqueda Dicotómica (Binaria) ya que el arreglo se mantiene ordenado
int buscarProducto(struct Producto inv[], int cantidad, int id_buscado) {
    int inicio = 0, fin = cantidad - 1, medio;
    while(inicio <= fin) {
        medio = (inicio + fin) / 2;
        if(inv[medio].nro_item == id_buscado) {
            return medio; // Encontrado
        } else if(inv[medio].nro_item < id_buscado) {
            inicio = medio + 1;
        } else {
            fin = medio - 1;
        }
    }
    return -1; // No encontrado
}

// Inserción manteniendo el ordenamiento (Insertion Sort like)
void insertarProducto(struct Producto inv[], int *cantidad, struct Producto nuevo) {
    if(*cantidad >= MAX_ITEMS) {
        printf("Error: Capacidad maxima alcanzada.\n");
        return;
    }
    
    int i = *cantidad - 1;
    // Desplazar elementos mayores hacia la derecha
    while(i >= 0 && inv[i].nro_item > nuevo.nro_item) {
        inv[i + 1] = inv[i];
        i--;
    }
    inv[i + 1] = nuevo;
    (*cantidad)++;
    printf("Producto %d insertado correctamente.\n", nuevo.nro_item);
}

Ejercicio 4: Control de Tránsito

20 inspectores con hasta 2000 actas cada uno. Registros: `nroActa, dominio, tipoInfraccion, idInspector`. Descargar a un soporte único, ordenar por tipoInfraccion e inspector, y mostrar subtotales y porcentajes (Corte de Control).

#include <stdio.h>

#define TOTAL_ACTAS 6 // Reducido para ejemplo

struct Acta {
    int nroActa;
    int tipoInfraccion;
    int idInspector;
};

// Ordenar por tipoInfraccion (1er criterio) y luego idInspector (2do criterio)
void ordenarActas(struct Acta actas[], int n) {
    struct Acta aux;
    for(int i = 0; i < n - 1; i++) {
        for(int j = 0; j < n - i - 1; j++) {
            if(actas[j].tipoInfraccion > actas[j+1].tipoInfraccion || 
              (actas[j].tipoInfraccion == actas[j+1].tipoInfraccion && actas[j].idInspector > actas[j+1].idInspector)) {
                aux = actas[j];
                actas[j] = actas[j+1];
                actas[j+1] = aux;
            }
        }
    }
}

// Logica de Corte de Control
void reporteInfracciones(struct Acta actas[], int n) {
    if (n == 0) return;
    
    int i = 0;
    int total_general = 0;
    
    while(i < n) {
        int tipo_actual = actas[i].tipoInfraccion;
        int total_tipo = 0;
        
        printf("\nInfraccion tipo %d:\n", tipo_actual);
        
        while(i < n && actas[i].tipoInfraccion == tipo_actual) {
            int inspector_actual = actas[i].idInspector;
            int actas_inspector = 0;
            
            while(i < n && actas[i].tipoInfraccion == tipo_actual && actas[i].idInspector == inspector_actual) {
                actas_inspector++;
                total_tipo++;
                total_general++;
                i++;
            }
            // En un caso real el porcentaje requiere 2 pasadas o calculo post-tipo,
            // aqui se muestra la cantidad para no complicar el bucle 1er anio.
            printf("  Inspector %d: %d actas\n", inspector_actual, actas_inspector);
        }
        printf("Total infracciones tipo %d: %d\n", tipo_actual, total_tipo);
    }
    printf("\nCantidad total de infracciones general: %d\n", total_general);
}

int main() {
    struct Acta sistema[TOTAL_ACTAS] = {
        {101, 2, 5},
        {102, 1, 3},
        {103, 2, 5},
        {104, 1, 3},
        {105, 1, 10},
        {106, 2, 4}
    };
    
    ordenarActas(sistema, TOTAL_ACTAS);
    reporteInfracciones(sistema, TOTAL_ACTAS);
    
    return 0;
}

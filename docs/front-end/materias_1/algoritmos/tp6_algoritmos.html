Ejercicio 1: Aplicación con TAD AUTO

Suponer creado e implementado el TAD AUTO para desarrollar un programa de aplicación que permita crear 2 autos y luego determinar cuál de los 2 es más caro y si ambos son del mismo año de fabricación. Utilizar la interfaz provista (crearAuto, verPrecio, verAnio, etc).

Se implementa la estructura interna del TAD y sus funciones básicas en el mismo archivo para que el código sea funcional, emulando el comportamiento de un TAD_Auto.h. Se respeta la restricción de no usar variables globales.

#include <stdio.h>
#include <string.h>

// =====================================================================
// IMPLEMENTACION DEL TAD AUTO (Simulando el archivo TAD_Auto.h)
// =====================================================================
struct Auto {
    char patente[8];
    char marca[31];
    char modelo[31];
    int anio;
    float precio;
};

void crearAuto(struct Auto *A, char pat[], char mar[], char mod[], int a, float pre) {
    strcpy(A->patente, pat);
    strcpy(A->marca, mar);
    strcpy(A->modelo, mod);
    A->anio = a;
    A->precio = pre;
}

float verPrecio(struct Auto A) {
    return A.precio;
}

int verAnio(struct Auto A) {
    return A.anio;
}

void verPatente(struct Auto A, char dest[]) {
    strcpy(dest, A.patente);
}

// =====================================================================
// APLICACION PRINCIPAL
// =====================================================================
int main() {
    struct Auto auto1, auto2;
    char pat1[8], pat2[8];

    // Utilizacion de la interfaz publica del TAD
    crearAuto(&auto1, "AB123CD", "Toyota", "Corolla", 2020, 15000000.0);
    crearAuto(&auto2, "EF456GH", "Ford", "Focus", 2020, 14500000.0);

    printf("--- COMPARATIVA DE AUTOS ---\n");
    
    // Determinacion del auto mas caro
    if (verPrecio(auto1) > verPrecio(auto2)) {
        verPatente(auto1, pat1);
        printf("El auto mas caro es el de patente: %s (Precio: $%.2f)\n", pat1, verPrecio(auto1));
    } else if (verPrecio(auto2) > verPrecio(auto1)) {
        verPatente(auto2, pat2);
        printf("El auto mas caro es el de patente: %s (Precio: $%.2f)\n", pat2, verPrecio(auto2));
    } else {
        printf("Ambos autos tienen exactamente el mismo precio.\n");
    }

    // Determinacion de anio de fabricacion
    if (verAnio(auto1) == verAnio(auto2)) {
        printf("Ambos autos son del mismo anio de fabricacion (%d).\n", verAnio(auto1));
    } else {
        printf("Los autos son de distinto anio de fabricacion.\n");
    }

    return 0;
}

Ejercicio 2: TAD Departamento

Diseñar un TAD para la gestión de departamentos de distintas comodidades de un complejo turístico, la información se maneja con los siguientes atributos: cod_depto, ubicacion, capacidad, precio_diario, internet, tv y estado. Usando el TAD, se debe crear un departamento y verificar su disponibilidad, cambiar el estado, calcular precio por días y visualizar datos.

#include <stdio.h>
#include <string.h>

// =====================================================================
// ESPECIFICACION E IMPLEMENTACION DEL TAD DEPARTAMENTO
// =====================================================================
struct Departamento {
    int cod_depto;
    char ubicacion[50];
    int capacidad;
    float precio_diario;
    int internet; // 1: Si posee, 0: No posee
    int tv;       // 1: Si posee, 0: No posee
    int estado;   // 0: Disponible, 1: Ocupado, 2: Mantenimiento
};

void crearDepto(struct Departamento *D, int cod, char ub[], int cap, float precio, int net, int tele) {
    D->cod_depto = cod;
    strcpy(D->ubicacion, ub);
    D->capacidad = cap;
    D->precio_diario = precio;
    D->internet = net;
    D->tv = tele;
    D->estado = 0; // Por defecto se crea en estado Disponible (0)
}

int verificarDisponibilidad(struct Departamento D) {
    return (D.estado == 0); // Retorna 1 (Verdadero) si esta disponible
}

void cambiarEstado(struct Departamento *D, int nuevo_estado) {
    D->estado = nuevo_estado;
}

float calcularPrecioEstadia(struct Departamento D, int dias) {
    return D.precio_diario * dias;
}

void mostrarDepto(struct Departamento D) {
    printf("--- Ficha del Departamento %d ---\n", D.cod_depto);
    printf("Ubicacion: %s | Capacidad Max: %d personas\n", D.ubicacion, D.capacidad);
    printf("Precio Diario: $%.2f | Internet: %s | TV: %s\n", D.precio_diario, D.internet ? "Si" : "No", D.tv ? "Si" : "No");
    
    printf("Estado actual: ");
    if (D.estado == 0) printf("Disponible\n");
    else if (D.estado == 1) printf("Ocupado\n");
    else printf("En Mantenimiento\n");
}

// =====================================================================
// APLICACION PRINCIPAL
// =====================================================================
int main() {
    struct Departamento depto1;
    int dias_reserva = 5;

    // 1. Crear departamento
    crearDepto(&depto1, 101, "Mar del Plata - Frente al Mar", 4, 25000.0, 1, 1);
    
    // 2. Visualizacion inicial
    mostrarDepto(depto1);

    // 3. Verificar disponibilidad, cambiar estado y calcular precio
    if (verificarDisponibilidad(depto1)) {
        printf("\nEl departamento se encuentra DISPONIBLE. Procediendo a generar reserva...\n");
        cambiarEstado(&depto1, 1); // Lo marcamos como Ocupado
        
        float total = calcularPrecioEstadia(depto1, dias_reserva);
        printf("Reserva confirmada por %d dias. Total a abonar: $%.2f\n", dias_reserva, total);
    } else {
        printf("\nLo sentimos, el departamento no se encuentra disponible actualmente.\n");
    }
    
    // 4. Visualizacion final para comprobar el cambio
    printf("\nEstado actualizado post-reserva:\n");
    mostrarDepto(depto1);

    return 0;
}

Ejercicio 3: TAD Producto

Diseñar un TAD para la gestión de inventario de productos de un comercio. La información se maneja con los atributos: Cod, Marca, Tipo, precio_costo, QR, cantidad y Proveedor. Usando el TAD, se debe crear un producto y verificar cantidad disponible, cambiar cantidad, visualizar código QR y mostrar datos del producto.

#include <stdio.h>
#include <string.h>

// =====================================================================
// ESPECIFICACION E IMPLEMENTACION DEL TAD PRODUCTO
// =====================================================================
struct Producto {
    int cod;
    char marca[30];
    char tipo[30];
    float precio_costo;
    char qr[50];
    int cantidad;
    char proveedor[30];
};

void crearProducto(struct Producto *P, int c, char mar[], char tip[], float pre, char cod_qr[], int cant, char prov[]) {
    P->cod = c;
    strcpy(P->marca, mar);
    strcpy(P->tipo, tip);
    P->precio_costo = pre;
    strcpy(P->qr, cod_qr);
    P->cantidad = cant;
    strcpy(P->proveedor, prov);
}

int verificarCantidad(struct Producto P) {
    return P.cantidad;
}

void cambiarCantidad(struct Producto *P, int nueva_cantidad) {
    if (nueva_cantidad >= 0) {
        P->cantidad = nueva_cantidad;
    } else {
        printf("Error de inventario: La cantidad no puede ser menor a cero.\n");
    }
}

void visualizarQR(struct Producto P) {
    printf("\n=== ESCANEO DE CODIGO QR ===\n");
    printf("  [ %s ]  \n", P.qr);
    printf("============================\n");
}

void mostrarProducto(struct Producto P) {
    printf("\n--- Detalle del Producto #%d ---\n", P.cod);
    printf("Tipo: %s | Marca: %s\n", P.tipo, P.marca);
    printf("Proveedor: %s | Precio de Costo: $%.2f\n", P.proveedor, P.precio_costo);
    printf("Stock Actual: %d unidades\n", P.cantidad);
}

// =====================================================================
// APLICACION PRINCIPAL
// =====================================================================
int main() {
    struct Producto prod1;

    // 1. Crear el producto usando la interfaz del TAD
    crearProducto(&prod1, 5501, "Logitech", "Mouse Inalambrico", 12500.50, "QR-LOGI-5501-MX", 15, "TechSupplies SA");
    
    // 2. Mostrar datos del producto recien creado
    mostrarProducto(prod1);
    
    // 3. Simular visualizacion de codigo QR
    visualizarQR(prod1);
    
    // 4. Verificar cantidad y procesar una venta o ajuste
    printf("\nStock reportado por el sistema: %d unidades.\n", verificarCantidad(prod1));
    printf("Se registra la venta de 5 unidades...\n");
    
    // 5. Cambiar cantidad
    cambiarCantidad(&prod1, verificarCantidad(prod1) - 5);
    
    printf("\nEstado actualizado del inventario:\n");
    mostrarProducto(prod1);

    return 0;
}

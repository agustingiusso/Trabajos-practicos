Ejercicio 1: Promedio entero de una comisión de 40 alumnos

Se desea calcular el promedio de notas de una comisión de 40 alumnos. Imprimir el resultado como un número entero (si hay parte decimal, eliminarla).

#include <stdio.h>

int main()
{
    int i, promedio_entero = 0;
    float nota = 0, suma = 0;
    const int TOTAL_ALUMNOS = 40;

    for (i = 1; i <= TOTAL_ALUMNOS; i++)
    {
        printf("Ingrese la nota del alumno %d: ", i);
        scanf("%f", &nota);
        suma = suma + nota;
    }

    // Conversión explícita a int para truncar la parte decimal:
    promedio_entero = (int)(suma / TOTAL_ALUMNOS);

    printf("\nEl promedio entero de la comision es: %d\n", promedio_entero);

    return 0;
}

Ejercicio 2: Porcentaje de aprobados y nota promedio de aprobación

Se desea calcular e imprimir el porcentaje de alumnos aprobados en una comisión de 40 alumnos y la nota promedio de aprobación. El porcentaje debe calcularse con 2 dígitos decimales y la nota promedio debe ser entera y redondearse para abajo.

#include <stdio.h>

int main()
{
    int i, aprobados = 0, promedio_aprobacion = 0;
    float nota = 0, suma_aprobados = 0, porcentaje_aprobados = 0;
    const int TOTAL_ALUMNOS = 40;

    for (i = 1; i <= TOTAL_ALUMNOS; i++)
    {
        printf("Ingrese la nota del alumno %d: ", i);
        scanf("%f", &nota);

        // Considerando aprobación con nota >= 6:
        if (nota >= 6.0)
        {
            aprobados++;
            suma_aprobados = suma_aprobados + nota;
        }
    }

    porcentaje_aprobados = (aprobados * 100.0) / TOTAL_ALUMNOS;

    printf("\nPorcentaje de alumnos aprobados: %.2f%%\n", porcentaje_aprobados);

    if (aprobados > 0)
    {
        promedio_aprobacion = (int)(suma_aprobados / aprobados);
        printf("Nota promedio de aprobacion (redondeada hacia abajo): %d\n", promedio_aprobacion);
    }
    else
    {
        printf("No se registraron alumnos aprobados en la comision.\n");
    }

    return 0;
}

Ejercicio 3: Encuesta demográfica en tres ciudades

Se realiza una encuesta en tres ciudades de un mismo distrito para determinar cuál de ellas tiene el mayor porcentaje de chicos menores de 5 años. Imprimir un mensaje indicándolo. Se sabe que la ciudad 1 tienen 1000 chicos menores de 10 años, la ciudad 2 tiene 1500 chicos menores de 10 años y la ciudad 3 tiene 1530 chicos menores de 10 años.

#include <stdio.h>

int main()
{
    int i, edad = 0, menores_5 = 0, ciudad_mayor = 0;
    int poblacion[3] = {1000, 1500, 1530};
    int c;
    float porcentaje = 0, max_porcentaje = -1.0;

    for (c = 0; c < 3; c++)
    {
        menores_5 = 0;
        printf("\n--- Relevamiento Ciudad %d (%d chicos menores de 10 anios) ---\n", c + 1, poblacion[c]);

        for (i = 1; i <= poblacion[c]; i++)
        {
            printf("Edad del chico %d (ciudad %d): ", i, c + 1);
            scanf("%d", &edad);

            if (edad < 5)
            {
                menores_5++;
            }
        }

        porcentaje = (menores_5 * 100.0) / poblacion[c];
        printf("Porcentaje de menores de 5 anios en Ciudad %d: %.2f%%\n", c + 1, porcentaje);

        if (porcentaje > max_porcentaje)
        {
            max_porcentaje = porcentaje;
            ciudad_mayor = c + 1;
        }
    }

    printf("\nLa Ciudad %d tiene el mayor porcentaje de chicos menores de 5 anios (%.2f%%).\n",
           ciudad_mayor, max_porcentaje);

    return 0;
}

Ejercicio 8: Potencias de P desde la posición Q hasta R

Implementar un programa que lea tres números P, Q, R e imprima las potencias de P empezando desde la que está en la posición Q hasta la R.

#include <stdio.h>
#include <math.h>

int main()
{
    float P = 0;
    int Q = 0, R = 0, exp;

    printf("Ingrese la base (P): ");
    scanf("%f", &P);
    printf("Ingrese el exponente inicial (Q) y final (R): ");
    scanf("%d %d", &Q, &R);

    if (Q > R)
    {
        printf("Error: La posicion inicial Q debe ser menor o igual a R.\n");
        return 1;

    }

    return 0;
}

Ejercicio 11: Raíces reales de una ecuación de segundo grado

Implementar un programa que calcule e imprima las raíces reales de una ecuación de segundo grado o un cartel indicando que no existen tales raíces.

#include <stdio.h>
#include <math.h>

int main()
{
    float a = 0, b = 0, c = 0, discriminante = 0, x1 = 0, x2 = 0;

    printf("Ingrese los coeficientes a, b y c de la ecuacion cuadratica: ");
    scanf("%f %f %f", &a, &b, &c);

    if (a == 0)
    {
        printf("El coeficiente 'a' no puede ser 0 en una ecuacion de segundo grado.\n");
        return 1;
    }

    discriminante = (b * b) - (4.0 * a * c);

    if (discriminante > 0)
    {
        x1 = (-b + sqrt(discriminante)) / (2.0 * a);
        x2 = (-b - sqrt(discriminante)) / (2.0 * a);
        printf("Raices reales distintas: x1 = %.2f | x2 = %.2f\n", x1, x2);
    }
    else if (discriminante == 0)
    {
        x1 = -b / (2.0 * a);
        printf("Raiz real doble: x1 = x2 = %.2f\n", x1);
    }
    else
    {
        printf("No existen raices reales para los coeficientes ingresados.\n");
    }

    return 0;
}

Ejercicio 13: Relevamiento simultáneo de 15 encuestadores en 3 ciudades

Quince encuestadores se reparten entre 3 ciudades para relevar entre la población si considera que cierto producto es bueno, malo, regular, o no lo conoce. El relevamiento se lleva a cabo simultáneamente en las 3 ciudades. Cada encuestador entrevista una cantidad fija de personas (a ser leída) y a medida que obtienen sus respuestas las envían junto con el código de la ciudad a la oficina central por medio de un dispositivo inalámbrico. En dicha oficina, la información es recibida por un programa que debe calcular e informar para cada ciudad el porcentaje de su población que ubica al producto en cada categoría.

#include <stdio.h>
#include <ctype.h>

int main()
{
    int cant_por_encuestador = 0, total_envios = 0, i, c, ciudad = 0;
    char opinion;
    int bueno[3] = {0, 0, 0};
    int malo[3] = {0, 0, 0};
    int regular[3] = {0, 0, 0};
    int no_conoce[3] = {0, 0, 0};
    int total_ciudad[3] = {0, 0, 0};

    printf("Ingrese la cantidad fija de personas entrevistadas por cada encuestador: ");
    scanf("%d", &cant_por_encuestador);

    total_envios = 15 * cant_por_encuestador;

    for (i = 1; i <= total_envios; i++)
    {
        printf("\n--- Envio #%d de %d ---\n", i, total_envios);
        printf("Codigo de ciudad (1, 2 o 3): ");
        scanf("%d", &ciudad);
        printf("Opinion (B: Bueno, M: Malo, R: Regular, N: No conoce): ");
        scanf(" %c", &opinion);
        opinion = toupper(opinion);

        if (ciudad >= 1 && ciudad <= 3)
        {
            total_ciudad[ciudad - 1]++;
            switch (opinion)
            {
            case 'B':
                bueno[ciudad - 1]++;
                break;
            case 'M':
                malo[ciudad - 1]++;
                break;
            case 'R':
                regular[ciudad - 1]++;
                break;
            case 'N':
                no_conoce[ciudad - 1]++;
                break;
            default:
                printf("Opinion no valida.\n");
                break;
            }
        }
    }

    printf("\n============= INFORME OFICINA CENTRAL =============\n");
    for (c = 0; c < 3; c++)
    {
        printf("\nCiudad %d (Total entrevistados: %d):\n", c + 1, total_ciudad[c]);
        if (total_ciudad[c] > 0)
        {
            printf("  - Bueno:      %.2f%%\n", (bueno[c] * 100.0) / total_ciudad[c]);
            printf("  - Malo:       %.2f%%\n", (malo[c] * 100.0) / total_ciudad[c]);
            printf("  - Regular:    %.2f%%\n", (regular[c] * 100.0) / total_ciudad[c]);
            printf("  - No conoce:  %.2f%%\n", (no_conoce[c] * 100.0) / total_ciudad[c]);
        }
        else
        {
            printf("  Sin datos registrados para esta ciudad.\n");
        }
    }

    return 0;
}

Ejercicio 1: Promedios de 5 comisiones de Matemática y promedio general

En una institución educativa de nivel terciario, los docentes de la asignatura Matemática del 3er. año de todas las carreras quieren determinar la nota promedio para cada una de las 5 comisiones y el promedio General. Para lograrlo, se le pide el examen a cada alumno para registrar la nota. Todos los promedios se calculan sobre la cantidad de alumnos presentes que tenían el examen consigo al momento de hacer el cálculo. Imprimir los resultados redondeados al entero más próximo.

#include <stdio.h>
#include <math.h>

int main()
{
    int comision, cant_comision = 0, total_general_alumnos = 0;
    float nota = 0, suma_comision = 0, suma_general = 0;
    int prom_comision_redondeado = 0, prom_general_redondeado = 0;

    for (comision = 1; comision <= 5; comision++)
    {
        suma_comision = 0;
        cant_comision = 0;

        printf("\n--- Comision %d (Ingrese nota negativa para finalizar la comision) ---\n", comision);
        printf("Nota del alumno presente con examen: ");
        scanf("%f", &nota);

        while (nota >= 0)
        {
            suma_comision += nota;
            cant_comision++;

            printf("Nota del siguiente alumno (negativo para cortar): ");
            scanf("%f", &nota);
        }

        if (cant_comision > 0)
        {
            prom_comision_redondeado = (int)round(suma_comision / cant_comision);
            printf("Promedio Comision %d (redondeado): %d\n", comision, prom_comision_redondeado);
            suma_general += suma_comision;
            total_general_alumnos += cant_comision;
        }
        else
        {
            printf("Comision %d sin alumnos con examen presentes.\n", comision);
        }
    }

    if (total_general_alumnos > 0)
    {
        prom_general_redondeado = (int)round(suma_general / total_general_alumnos);
        printf("\n==================================================\n");
        printf("PROMEDIO GENERAL DE LAS 5 COMISIONES: %d\n", prom_general_redondeado);
        printf("==================================================\n");
    }

    return 0;
}

Ejercicio 2: Clasificación y numeración correlativa de cartas por sección

#include <stdio.h>

int main()
{

  • Análisis del bucle a): Al comenzar con I = 0, se asigna J = 0. En el ciclo interno se cumple 0 * 0 < 100, imprime 0 y calcula J = 0 * 2 = 0. Como J nunca cambia de 0, entra en un bucle infinito imprimiendo 0 indefinidamente.

    • En la primera entrada al bucle externo (I = 1, X = 1), el bucle interno se ejecuta 4 veces:
      1) 1 * 1 < 800 → X = 11, I = 3
      2) 11 * 3 = 33 < 800 → X = 41, I = 5
      3) 41 * 5 = 205 < 800 → X = 91, I = 7
      4) 91 * 7 = 637 < 800 → X = 161, I = 9 (sale del ciclo interno porque 161 * 9 = 1449 >= 800).
    • Al terminar la 1ra vuelta externa, I pasa a 11. Desde I = 11 hasta I = 79 (inclusive, subiendo de 2 en 2), el bucle externo realiza 35 iteraciones más donde la condición interna X * I < 800 da siempre Falso, hasta que I llega a 81 y finaliza.

Ejercicio 14: Conversión de Repetir Mientras a Repetir Para

En los segmentos de código del ejercicio 13, ¿podría reemplazarse el Repetir Mientras por Repetir Para? De ser así, ¿cómo quedaría el código dentro de cada bucle?

Respuesta Teórica:

  • En Pseudocódigo estricto de la cátedra: La estructura Repetir Para es de repetición fija con paso constante sumado al índice y requiere conocer de antemano el límite de iteraciones sin alterar el índice manualmente dentro del cuerpo. Por lo tanto, en el bucle a) el ciclo externo sí equivale a un Repetir Para I := 1, 3, 1, pero el interno multiplica J := J * 2 (progresión geométrica, no paso constante). En el bucle b), la variable de control I es modificada también adentro del ciclo anidado, lo cual viola la regla de no alterar el índice de un Repetir Para.
  • En Lenguaje C: Sí es posible reemplazar ambos while por sentencias for (inicialización; condición; actualización) gracias a la flexibilidad del encabezado del for en C:
#include <stdio.h>

int main()
{
    int I, J, X;

    // Equivalente con for en C del segmento A (para I >= 1):
    printf("--- Segmento A con for ---\n");
    for (I = 1; I * I < 10; I++)
    {
        for (J = I; J * J < 100; J *= 2)
        {
            printf("%d ", I + J);
        }
    }
    printf("\n");

    // Equivalente con for en C del segmento B:
    printf("\n--- Segmento B con for ---\n");
    X = 1;
    for (I = 1; I < 80; I += 2)
    {
        for (; X * I < 800; I += 2)
        {
            X = X + I * 10;
        }
    }
    printf("Salida -> X = %d, I = %d\n", X, I);

    return 0;
}

                if (i >= 2 && i + 2 < len)
                {
                    if (palabra[i] == 'c' && palabra[i + 1] == 'a' && palabra[i + 2] == 's')
                    {
                        tiene_cas_3 = 1;
                    }
                }
            }

            if (tiene_cas_3) cont_cas_manual++;
            if (tiene_c && tiene_a && tiene_s) cont_letras_manual++;

            // =========================================================
            // VERSIÓN 2: Usando funciones primitivas (strlen, strstr, strchr)
            // =========================================================
            if ((int)strlen(palabra) >= 5 && strstr(palabra + 2, "cas") != NULL)
            {
                cont_cas_prim++;
            }

            if (strchr(palabra, 'c') != NULL &&
                strchr(palabra, 'a') != NULL &&
                strchr(palabra, 's') != NULL)
            {
                cont_letras_prim++;
            }

            len = 0; // Reinicio el buffer para la siguiente palabra
        }

    } while (ch != '.');

    printf("\n============= RESULTADOS DEL ANALISIS DE TEXTO =============\n");
    printf("Total de palabras procesadas: %d\n", total_palabras);
    printf("Longitud de la palabra mas larga: %d caracteres\n\n", max_longitud);

    printf("--- Version 1 (Sin funciones primitivas) ---\n");
    printf("Palabras con \"cas\" desde la 3ra letra: %d\n", cont_cas_manual);
    printf("Palabras con 'c', 'a' y 's': %d (%.2f%%)\n\n",
           cont_letras_manual, total_palabras > 0 ? (cont_letras_manual * 100.0) / total_palabras : 0.0);

    printf("--- Version 2 (Con funciones de string.h) ---\n");
    printf("Palabras con \"cas\" desde la 3ra letra: %d\n", cont_cas_prim);
    printf("Palabras con 'c', 'a' y 's': %d (%.2f%%)\n",
           cont_letras_prim, total_palabras > 0 ? (cont_letras_prim * 100.0) / total_palabras : 0.0);

    return 0;
}
